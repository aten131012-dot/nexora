-- NEXORA HUB LOADER
local Players = game:GetService("Players")
local TS      = game:GetService("TweenService")
local LP      = Players.LocalPlayer

local CFG = {
    DISCORD = "https://discord.gg/vHRMNpDsQ4",
}
local SFX = {
    click = "rbxasset://sounds/button.wav",
    open  = "rbxasset://sounds/switch.wav",
    boom  = "rbxasset://sounds/impact_water.mp3",
}
local Scripts = {
    [142823291] = "https://raw.githubusercontent.com/aten131012-dot/Obf/refs/heads/main/MM2.lua",

local LANG  = "TH"
local BLACK = Color3.new(0, 0, 0)
local WHITE = Color3.new(1, 1, 1)

-- ===== HELPERS =====
local function new(c, p, par)
    local i = Instance.new(c)
    for k, v in pairs(p) do i[k] = v end
    i.Parent = par
    return i
end
local function tw(o, t, p, s, d)
    local x = TS:Create(o, TweenInfo.new(t, s or Enum.EasingStyle.Quad, d or Enum.EasingDirection.Out), p)
    x:Play()
    return x
end
local function snd(id, vol)
    if not id or id == "" then return end
    local s = Instance.new("Sound")
    s.SoundId, s.Volume = id, vol or 0.6
    s.Parent = game:GetService("SoundService")
    s:Play()
    game:GetService("Debris"):AddItem(s, 6)
end
local function mkGui(name, order)
    pcall(function()
        local old = (gethui and gethui() or game:GetService("CoreGui")):FindFirstChild(name)
        if old then old:Destroy() end
    end)
    local g = Instance.new("ScreenGui")
    g.Name, g.ResetOnSpawn, g.IgnoreGuiInset, g.DisplayOrder = name, false, true, order
    g.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
    local ok = pcall(function() g.Parent = (gethui and gethui()) or game:GetService("CoreGui") end)
    if not ok or not g.Parent then g.Parent = LP:WaitForChild("PlayerGui") end
    return g
end

-- ===== NOTICE POPUP =====
local noticeOpen = false
local function ShowNotice()
    if noticeOpen then return end
    noticeOpen = true
    snd(SFX.open, 0.8)

    local T = {
        EN = {
            msg     = "This map has no script yet.\nFollow updates on our Discord.",
            copy    = "COPY DISCORD LINK",
            copied  = "COPIED!",
            close   = "CLOSE",
            langBtn = "TH",
        },
        TH = {
            msg     = "แมพนี้ยังไม่มีสคริปต์\nติดตามข่าวสารได้ทางดิสคอส",
            copy    = "คัดลอกลิ้งค์",
            copied  = "คัดลอกแล้ว!",
            close   = "ปิด",
            langBtn = "EN",
        },
    }

    local gui  = mkGui("NEXORA_NOTICE", 999)
    local vs   = workspace.CurrentCamera.ViewportSize
    local W, H = 320, 196

    -- dim background
    new("Frame", {
        Size = UDim2.fromScale(1, 1),
        BackgroundColor3 = BLACK,
        BackgroundTransparency = 0.45,
        BorderSizePixel = 0,
        ZIndex = 2,
    }, gui)

    -- main panel
    local panel = new("Frame", {
        AnchorPoint      = Vector2.new(0.5, 0.5),
        Position         = UDim2.fromScale(0.5, 0.5),
        Size             = UDim2.fromOffset(W, 4),
        BackgroundColor3 = Color3.fromRGB(12, 12, 16),
        BorderSizePixel  = 0,
        ClipsDescendants = true,
        ZIndex           = 3,
    }, gui)
    new("UIScale",  {Scale = math.min(1, vs.X * 0.92 / W, vs.Y * 0.92 / H)}, panel)
    new("UICorner", {CornerRadius = UDim.new(0, 14)}, panel)
    new("UIStroke", {Color = Color3.fromRGB(55, 55, 75), Thickness = 1}, panel)

    -- top bar
    local topBar = new("Frame", {
        Size             = UDim2.fromOffset(W, 38),
        BackgroundColor3 = Color3.fromRGB(18, 18, 26),
        BorderSizePixel  = 0,
        ZIndex           = 4,
    }, panel)
    new("UICorner", {CornerRadius = UDim.new(0, 14)}, topBar)

    -- title
    new("TextLabel", {
        Position           = UDim2.fromOffset(14, 9),
        Size               = UDim2.fromOffset(200, 20),
        BackgroundTransparency = 1,
        Text               = "NEXORA HUB",
        TextColor3         = WHITE,
        Font               = Enum.Font.GothamBlack,
        TextSize           = 14,
        TextXAlignment     = Enum.TextXAlignment.Left,
        ZIndex             = 5,
    }, topBar)

    -- lang toggle
    local langBtn = new("TextButton", {
        Position         = UDim2.fromOffset(W - 54, 9),
        Size             = UDim2.fromOffset(44, 20),
        BackgroundColor3 = Color3.fromRGB(35, 35, 50),
        TextColor3       = Color3.fromRGB(180, 180, 210),
        Font             = Enum.Font.GothamBold,
        TextSize         = 11,
        BorderSizePixel  = 0,
        ZIndex           = 5,
    }, topBar)
    new("UICorner", {CornerRadius = UDim.new(0, 5)}, langBtn)
    new("UIStroke", {Color = Color3.fromRGB(70, 70, 100), Thickness = 1}, langBtn)

    -- message label
    local msgLabel = new("TextLabel", {
        Position           = UDim2.fromOffset(14, 48),
        Size               = UDim2.fromOffset(W - 28, 68),
        BackgroundTransparency = 1,
        TextColor3         = Color3.fromRGB(210, 210, 225),
        Font               = Enum.Font.Gotham,
        TextSize           = 13,
        TextWrapped        = true,
        TextXAlignment     = Enum.TextXAlignment.Center,
        TextYAlignment     = Enum.TextYAlignment.Top,
        ZIndex             = 4,
    }, panel)

    -- copy button
    local copyBtn = new("TextButton", {
        Position         = UDim2.fromOffset(14, 122),
        Size             = UDim2.fromOffset(W - 28, 32),
        BackgroundColor3 = Color3.fromRGB(38, 38, 58),
        TextColor3       = WHITE,
        Font             = Enum.Font.GothamBold,
        TextSize         = 12,
        BorderSizePixel  = 0,
        ZIndex           = 4,
    }, panel)
    new("UICorner", {CornerRadius = UDim.new(0, 8)}, copyBtn)
    new("UIStroke", {Color = Color3.fromRGB(80, 80, 120), Thickness = 1}, copyBtn)

    -- close button
    local closeBtn = new("TextButton", {
        Position         = UDim2.fromOffset(W / 2 - 52, 162),
        Size             = UDim2.fromOffset(104, 26),
        BackgroundColor3 = Color3.fromRGB(50, 18, 18),
        TextColor3       = Color3.fromRGB(240, 100, 100),
        Font             = Enum.Font.GothamBold,
        TextSize         = 12,
        BorderSizePixel  = 0,
        ZIndex           = 4,
    }, panel)
    new("UICorner", {CornerRadius = UDim.new(0, 7)}, closeBtn)
    new("UIStroke", {Color = Color3.fromRGB(120, 40, 40), Thickness = 1}, closeBtn)

    -- apply language
    local function applyLang()
        local L = T[LANG]
        langBtn.Text  = L.langBtn
        msgLabel.Text = L.msg
        copyBtn.Text  = L.copy
        closeBtn.Text = L.close
    end
    applyLang()

    langBtn.MouseButton1Click:Connect(function()
        snd(SFX.click, 0.5)
        LANG = (LANG == "EN") and "TH" or "EN"
        applyLang()
    end)

    local busy = false
    copyBtn.MouseButton1Click:Connect(function()
        if busy then return end
        busy = true
        snd(SFX.click, 0.8)
        pcall(function() (setclipboard or toclipboard)(CFG.DISCORD) end)
        copyBtn.Text             = T[LANG].copied
        copyBtn.BackgroundColor3 = Color3.fromRGB(28, 108, 65)
        task.wait(1.5)
        copyBtn.Text             = T[LANG].copy
        copyBtn.BackgroundColor3 = Color3.fromRGB(38, 38, 58)
        busy = false
    end)

    closeBtn.MouseButton1Click:Connect(function()
        snd(SFX.boom, 0.8)
        tw(panel, 0.3, {Size = UDim2.fromOffset(W, 0)}, Enum.EasingStyle.Back, Enum.EasingDirection.In)
        task.wait(0.35)
        gui:Destroy()
        noticeOpen = false
    end)

    -- open animation
    tw(panel, 0.45, {Size = UDim2.fromOffset(W, H)}, Enum.EasingStyle.Back)
end

-- ===== ENTRY POINT =====
local url = Scripts[game.PlaceId] or Scripts[game.GameId]

if url and tostring(url):find("^http") then
    local ok, err = pcall(function() loadstring(game:HttpGet(url))() end)
    if not ok then warn("[NEXORA] โหลดสคริปต์ไม่สำเร็จ: " .. tostring(err)) end
else
    task.spawn(ShowNotice)
end

-- ===== HELPERS =====
local function new(c, p, par)
    local i = Instance.new(c)
    for k, v in pairs(p) do i[k] = v end
    i.Parent = par
    return i
end
local function tw(o, t, p, s, d)
    local x = TS:Create(o, TweenInfo.new(t, s or Enum.EasingStyle.Quad, d or Enum.EasingDirection.Out), p)
    x:Play()
    return x
end
local function snd(id, vol)
    if not id or id == "" then return end
    local s = Instance.new("Sound")
    s.SoundId, s.Volume = id, vol or 0.6
    s.Parent = game:GetService("SoundService")
    s:Play()
    game:GetService("Debris"):AddItem(s, 6)
end
local function mkGui(name, order)
    pcall(function()
        local old = (gethui and gethui() or game:GetService("CoreGui")):FindFirstChild(name)
        if old then old:Destroy() end
    end)
    local g = Instance.new("ScreenGui")
    g.Name, g.ResetOnSpawn, g.IgnoreGuiInset, g.DisplayOrder = name, false, true, order
    g.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
    local ok = pcall(function() g.Parent = (gethui and gethui()) or game:GetService("CoreGui") end)
    if not ok or not g.Parent then g.Parent = LP:WaitForChild("PlayerGui") end
    return g
end

-- ===== MINI BADGE (มุมซ้าย ลอยได้ ดาวหมุน) =====
function ShowMiniBadge()
    local gui = mkGui("NEXORA_BADGE", 997)

    -- draggable badge
    local badge = new("Frame", {
        AnchorPoint = Vector2.new(0, 0.5),
        Position    = UDim2.fromOffset(10, 60),
        Size        = UDim2.fromOffset(140, 36),
        BackgroundColor3 = Color3.fromRGB(14,14,18),
        BorderSizePixel  = 0,
        ZIndex = 3,
        Active = true,
    }, gui)
    new("UICorner", {CornerRadius = UDim.new(0,8)}, badge)
    new("UIStroke", {Color = Color3.fromRGB(60,60,90), Thickness = 1}, badge)

    -- logo
    new("ImageLabel", {
        Position = UDim2.fromOffset(6, 7),
        Size     = UDim2.fromOffset(22, 22),
        BackgroundTransparency = 1,
        Image    = "rbxassetid://86383017875768",
        ZIndex   = 4,
    }, badge)

    new("TextLabel", {
        Position = UDim2.fromOffset(32, 9),
        Size     = UDim2.fromOffset(100, 18),
        BackgroundTransparency = 1,
        Text     = "NEXORA HUB",
        TextColor3 = WHITE,
        Font     = Enum.Font.GothamBlack,
        TextSize = 11,
        TextXAlignment = Enum.TextXAlignment.Left,
        ZIndex   = 4,
    }, badge)

    -- stars orbiting (SVG-style frame dots)
    local starCanvas = new("Frame", {
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position    = UDim2.fromScale(0.5, 0.5),
        Size        = UDim2.fromOffset(160, 56),
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        ZIndex = 2,
    }, badge)

    local stars = {}
    for i = 1, 6 do
        local dot = new("Frame", {
            Size  = UDim2.fromOffset(4, 4),
            BackgroundColor3 = Color3.fromRGB(180 + math.random(-30,30), 180, 255),
            BorderSizePixel  = 0,
            ZIndex = 2,
        }, starCanvas)
        new("UICorner", {CornerRadius = UDim.new(1,0)}, dot)
        stars[i] = {dot = dot, angle = (i-1)*(math.pi*2/6), radius = 60 + math.random(-6,6), speed = 0.6 + math.random()*0.4}
    end

    -- black hole effect ring behind badge
    local bhRing = new("Frame", {
        AnchorPoint = Vector2.new(0, 0.5),
        Position    = UDim2.fromOffset(-8, 18),
        Size        = UDim2.fromOffset(156, 52),
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        ZIndex = 1,
    }, badge)
    new("UICorner", {CornerRadius = UDim.new(1,0)}, bhRing)
    local bhStroke = new("UIStroke", {
        Color = Color3.fromRGB(100, 30, 200),
        Thickness = 2,
        Transparency = 0.4,
    }, bhRing)
    TS:Create(bhStroke, TweenInfo.new(1.4, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut, -1, true),
        {Transparency = 0.85, Color = Color3.fromRGB(160, 80, 255)}):Play()

    -- animate stars
    local alive = {on = true}
    task.spawn(function()
        local t = 0
        while alive.on and badge.Parent do
            t += task.wait()
            local cx, cy = 70, 18  -- center of starCanvas (half of 140,36)
            for _, s in ipairs(stars) do
                s.angle += s.speed * task.wait() * 0.5
                local px = cx + math.cos(s.angle) * (s.radius * 0.5) - 2
                local py = cy + math.sin(s.angle) * (s.radius * 0.18) - 2
                s.dot.Position = UDim2.fromOffset(px, py)
            end
        end
    end)

    -- drag logic
    local dragging, dragStart, posStart = false, nil, nil
    badge.InputBegan:Connect(function(inp)
        if inp.UserInputType == Enum.UserInputType.MouseButton1 or inp.UserInputType == Enum.UserInputType.Touch then
            dragging  = true
            dragStart = inp.Position
            posStart  = badge.Position
        end
    end)
    badge.InputEnded:Connect(function(inp)
        if inp.UserInputType == Enum.UserInputType.MouseButton1 or inp.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    game:GetService("UserInputService").InputChanged:Connect(function(inp)
        if dragging and (inp.UserInputType == Enum.UserInputType.MouseMovement or inp.UserInputType == Enum.UserInputType.Touch) then
            local delta = inp.Position - dragStart
            badge.Position = UDim2.fromOffset(
                posStart.X.Offset + delta.X,
                posStart.Y.Offset + delta.Y
            )
        end
    end)

    -- click to open notice
    local clickTime = 0
    badge.InputBegan:Connect(function(inp)
        if inp.UserInputType == Enum.UserInputType.MouseButton1 or inp.UserInputType == Enum.UserInputType.Touch then
            clickTime = tick()
        end
    end)
    badge.InputEnded:Connect(function(inp)
        if (inp.UserInputType == Enum.UserInputType.MouseButton1 or inp.UserInputType == Enum.UserInputType.Touch)
            and (tick() - clickTime < 0.25) and not dragging then
            snd(SFX.open, 0.6)
            alive.on = false
            gui:Destroy()
            ShowNotice()
        end
    end)
end
